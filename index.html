/* LEITOR DE PLANILHA ATUALIZADO: DUAS CARRETAS + SUPORTE A DATAS PASSA DAS */ function importarArquivo(e) { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(evt) { try { let data = evt.target.result; let workbook; if (typeof data === 'string') { workbook = XLSX.read(data, { type: 'binary' }); } else { let bytes = new Uint8Array(data); workbook = XLSX.read(bytes, { type: 'array' }); } let novos = []; let today = getTodayString(); // Percorre todas as abas da planilha (ex: Primeira Carreta, Segunda Carreta) workbook.SheetNames.forEach((sheetName) => { const worksheet = workbook.Sheets[sheetName]; const rawMatrix = XLSX.utils.sheet_to_json(worksheet, { header: 1, defval: "" }); if (!rawMatrix || rawMatrix.length === 0) return; let colIndexes = { id: -1, driver: -1, plate: -1, dest: -1, date: -1 }; for (let r = 0; r < rawMatrix.length; r++) { let row = rawMatrix[r]; if (!row || !row.length) continue; let rowStrArr = row.map(cell => String(cell).trim().toLowerCase()); let foundHeader = rowStrArr.some(c => c.includes('carga') || c.includes('veiculo') || c.includes('veículo') || c.includes('romaneio')); // Mapeamento dos cabeçalhos if (foundHeader && colIndexes.id === -1) { rowStrArr.forEach((cell, idx) => { if (cell.includes('data') || cell.includes('dia')) colIndexes.date = idx; if (cell.includes('carga') || cell.includes('romaneio') || cell.includes('erp') || cell === 'id') { if (colIndexes.id === -1) colIndexes.id = idx; } if (cell.includes('motorista') || cell.includes('nome') || cell.includes('condutor')) colIndexes.driver = idx; if (cell.includes('veiculo') || cell.includes('veículo') || cell.includes('placa') || cell.includes('carreta')) colIndexes.plate = idx; if (cell.includes('destino') || cell.includes('cidade') || cell.includes('uf')) colIndexes.dest = idx; }); continue; } // Processamento dos dados if (colIndexes.id !== -1) { let idVal = row[colIndexes.id] ? String(row[colIndexes.id]).trim() : ''; let driverVal = colIndexes.driver !== -1 && row[colIndexes.driver] ? String(row[colIndexes.driver]).trim() : '-'; let plateVal = colIndexes.plate !== -1 && row[colIndexes.plate] ? String(row[colIndexes.plate]).trim() : '-'; let destVal = colIndexes.dest !== -1 && row[colIndexes.dest] ? String(row[colIndexes.dest]).trim() : '-'; // Leitura e formatação da data let recordDate = today; if (colIndexes.date !== -1 && row[colIndexes.date]) { let rawDate = String(row[colIndexes.date]).trim(); // Se a data estiver no formato DD/MM/AAAA, converte para AAAA-MM-DD if (rawDate.includes('/')) { let parts = rawDate.split('/'); if (parts.length === 3) { recordDate = `${parts[2]}-${parts[1].padStart(2, '0')}-${parts[0].padStart(2, '0')}`; } } else if (rawDate.length === 10 && rawDate.includes('-')) { recordDate = rawDate; } } // Identifica a posição da carreta com base na aba ou coluna let carretaPos = sheetName.toLowerCase().includes('segunda') ? '2ª Carreta' : '1ª Carreta'; if (idVal && idVal !== '-' && idVal.length >= 3 && !isNaN(idVal.replace('.', '').replace(',', ''))) { if (!cs.some(x => String(x.id) === String(idVal))) { let item = { id: idVal, d: driverVal, p: plateVal, c: destVal, s: 'Separação', dock: '', loadStart: null, loadEnd: null, createdAt: recordDate, carreta: carretaPos }; cs.push(item); novos.push(item); } } } } }); if (novos.length > 0) { salvarDados(novos); alert(`✅ ${novos.length} carga(s) importada(s) com sucesso das abas processadas!`); } else { alert('⚠️ Nenhuma carga nova encontrada na planilha.'); } } catch (err) { console.error(err); alert('Erro ao ler a planilha. Verifique a estrutura dos arquivos.'); } e.target.value = ''; }; if (reader.readAsBinaryString) reader.readAsBinaryString(file); else reader.readAsArrayBuffer(file); }
